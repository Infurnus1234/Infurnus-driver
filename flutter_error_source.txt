class _OtpVerificationScreenState extends State<OtpVerificationScreen> {
  static const _digitCount = 6;

  final _focusNodes = List.generate(_digitCount, (_) => FocusNode());

  Timer? _timer;

  bool _hasError = false;

  bool get _complete => _otp.length == _digitCount;

  String _friendlyVerificationError(DriverSession session) {
    final contact = session.verificationContact ?? 'your account';
    final details = '${session.errorCode ?? ''} ${session.error ?? ''}'
        .toLowerCase();
    if (details.contains('expired')) {
      return 'That verification code expired for $contact. Request a new code.';
    }
    if (details.contains('invalid') ||
        details.contains('incorrect') ||
        details.contains('does not match')) {
      return 'That code does not match the one sent to $contact. Check it and try again.';
    }
    return 'We could not verify the code for $contact. Check it and try again.';
  }

  @override
  void initState() {
    super.initState();
    _timer = Timer.periodic(const Duration(seconds: 1), (_) {
      if (mounted) setState(() {});
    });
    // Listen to each box and advance focus automatically.
    for (var i = 0; i < _digitCount; i++) {
      final idx = i;
      _controllers[idx].addListener(() => _onDigitChanged(idx));
    }
  }

  @override
  void dispose() {
    _timer?.cancel();
    for (final c in _controllers) {
      c.dispose();
    }
    for (final f in _focusNodes) {
      f.dispose();
    }
    super.dispose();
  }

  void _onDigitChanged(int idx) {
    if (!mounted) return;
    final text = _controllers[idx].text;
    if (text.isEmpty && idx > 0) {
      // User deleted the digit; move focus back to the previous box.
      _focusNodes[idx - 1].requestFocus();
    } else if (text.length == 1 && idx < _digitCount - 1) {
      _focusNodes[idx + 1].requestFocus();
    }
    // If the user pasted a full 6-digit string into the first box, spread it.
    if (text.length > 1 && idx == 0) {
      final digits = text.replaceAll(RegExp(r'\D'), '');
      for (var i = 0; i < _digitCount && i < digits.length; i++) {
        _controllers[i].text = digits[i];
        _controllers[i].selection = TextSelection.collapsed(offset: 1);
      }
      final last = (digits.length - 1).clamp(0, _digitCount - 1);
      _focusNodes[last].requestFocus();
    }
    setState(() => _hasError = false);
  }

  Future<void> _verify(DriverSession session) async {
    if (_submitting || session.busy || !_complete) return;
    setState(() {
      _submitting = true;
      _hasError = false;
    });
    try {
      final signup = session.signupId != null;
      final ok = await session.verify(_otp);
      if (mounted && ok && session.authenticated) {
        context.go(signup ? '/onboarding' : session.landingPath);
      } else if (mounted && !ok) {
        setState(() => _hasError = true);
        for (final c in _controllers) {
          c.clear();
        }
        _focusNodes.first.requestFocus();
      }
    } finally {
      if (mounted) setState(() => _submitting = false);
    }
  }

  final _identifier = TextEditingController();
  bool _rememberMe = true;
  bool _useEmail = false;
  bool _identifierAttempted = false;
  String? _clientValidationMessage;

  bool get _identifierIsValid {
    final input = _identifier.text.trim();
    if (input.isEmpty) return false;
    if (_useEmail || input.contains('@')) {
      return RegExp(r'^[^\s@]+@[^\s@]+\.[^\s@]+$').hasMatch(input);
    }
    return true;
  }

  String? get _emailValidationError {
    if (!_useEmail || !_identifierAttempted) return null;
    final input = _identifier.text.trim();
    if (input.isEmpty) return 'Enter your email address.';
    return RegExp(r'^[^\s@]+@[^\s@]+\.[^\s@]+$').hasMatch(input)
        ? null
        : 'Enter a valid email address.';
  }

  String? _loginError(DriverSession session) {
    if (session.error == null) return null;
    if (session.errorCode == 'INVALID_CREDENTIALS') {
      return 'Those sign-in details could not be verified. Check them and retry.';
    }
    return 'We couldn\'t sign you in. Check your connection and try again.';
  }

  void _validateAndSubmit(DriverSession session) async {
    setState(() {
      _identifierAttempted = true;
      _clientValidationMessage = null;
    });
    final rawInput = _identifier.text.trim();

    if (rawInput.isEmpty) {
      setState(() {
        _clientValidationMessage = _useEmail
            ? 'Please enter your email address.'
            : 'Please enter your phone number.';
      });
      return;
    }

    if (_useEmail || rawInput.contains('@')) {
      if (!RegExp(r'^[^\s@]+@[^\s@]+\.[^\s@]+$').hasMatch(rawInput)) return;
    } else {
      final digits = rawInput.replaceAll(RegExp(r'\D'), '');
      if (digits.length < 10) {
        setState(
          () => _clientValidationMessage =
              'Please enter a valid 10-digit mobile number.',
        );
        return;
      }
    }

    final fullIdentifier = rawInput;

    session.api.rememberMe = _rememberMe;
    var success = await session.login(fullIdentifier);
    if (!mounted) return;
    if (!success && session.errorCode == 'ACCOUNT_ROLE_REQUIRED') {
      final role = await _chooseExistingRole();
      if (!mounted || role == null) return;
      success = await session.login(fullIdentifier, null, role);
    }
    if (!mounted) return;
    if (success) {
      context.go('/otp');
    }
  }

class _LedgerScreenState extends State<LedgerScreen> {
  String _period = 'daily';
  bool _requesting = false;
  int _revision = 0;
  Future<void> _payout() async {
    if (_requesting) return;
    final session = context.read<DriverSession>();
    final request = PayoutRequest(session.api);
    final controller = TextEditingController();
    setState(() => _requesting = true);
    try {
      final pending = await request.pending();
      if (!mounted) return;
      if (pending != null) {
        controller.text = ((pending['amountPaise'] as num) / 100)
            .toStringAsFixed(2);
      }
      final form = GlobalKey<FormState>();
      final paise = await showDialog<int>(
        context: context,
        builder: (context) => AlertDialog(
          title: Text(
            pending == null ? 'Request payout' : 'Retry pending payout',
          ),
          content: Form(
            key: form,
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                const Text(
                  'Requires a verified bank account and sufficient wallet balance. Request acceptance does not confirm bank settlement.',
                ),
                TextFormField(
                  controller: controller,
                  readOnly: pending != null,
                  keyboardType: const TextInputType.numberWithOptions(
                    decimal: true,
                  ),
                  decoration: const InputDecoration(labelText: 'Amount (INR)'),
                  validator: (text) {
                    if (!RegExp(r'^\d+(\.\d{1,2})?$')
                        .hasMatch(text?.trim() ?? '')) {
                      return 'Enter a positive amount with up to 2 decimals';
                    }
                    final amount = double.tryParse(text!.trim());
                    return amount == null || amount <= 0 || amount > 99999999.99
                        ? 'Enter an amount within the supported limit'
                        : null;
                  },
                ),
              ],
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('Cancel'),
            ),
            FilledButton(
              onPressed: () {
                if (form.currentState!.validate()) {
                  Navigator.pop(
                    context,
                    (double.parse(controller.text.trim()) * 100).round(),
                  );
                }
              },
              child: const Text('Submit request'),
            ),
          ],
        ),
      );
      if (paise == null || !mounted) return;
      final result = await request.submit(paise);
      if (mounted) {
        setState(() => _revision++);
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              'Payout request recorded: ${result is Map ? result['status'] ?? 'pending' : 'pending'}. Check payout history for its status.',
            ),
          ),
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              e is ApiException ? e.message : 'Payout could not be confirmed. Retry to reconcile the same request.',
            ),
          ),
        );
      }
    } finally {
      Future<void>.delayed(const Duration(seconds: 1), controller.dispose);
      if (mounted) setState(() => _requesting = false);
    }
  }

        builder: (context, data) {

class ApiPanel extends StatefulWidget {
  final ApiService api;
  final String path;
  final Widget Function(BuildContext, dynamic) builder;
  final bool embedded;
  final Duration? refreshInterval;
  const ApiPanel({
    super.key,
    required this.api,

class _ApiPanelState extends State<ApiPanel> with WidgetsBindingObserver {
  Timer? _refresh;
  bool _active = true;
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    _active = state == AppLifecycleState.resumed;
    if (_active && widget.refreshInterval != null && !_busy) _load();
  }

  @override
  void dispose() {
    _refresh?.cancel();
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  dynamic _data;
  String? _error;
  bool _busy = false;
  int _generation = 0;
  @override
  void initState() {
    super.initState();
    _load();
    WidgetsBinding.instance.addObserver(this);
    if (widget.refreshInterval != null) {
      _refresh = Timer.periodic(widget.refreshInterval!, (_) {
        if (mounted &&
            _active &&
            !_busy &&
            (ModalRoute.of(context)?.isCurrent ?? false)) {
          _load();
        }
      });
    }
  }

  @override
  void didUpdateWidget(ApiPanel oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.path != widget.path || oldWidget.api != widget.api) {
      _data = null;
      _load();
    }
  }

  Future<void> _load() async {
    final generation = ++_generation;
    setState(() {
      _busy = true;
      _error = null;
    });
    try {
      final data = await widget.api.request('GET', widget.path);
      if (mounted && generation == _generation) setState(() => _data = data);
    } catch (e, stack) {
      developer.log(
        'API panel request failed: ${widget.path}',
        name: 'ApiPanel',
        error: e,
        stackTrace: stack,
      );
      if (mounted && generation == _generation) {
        setState(
          () => _error = 'We couldn\'t load this information. Please retry.',
        );
      }
    } finally {
      if (mounted && generation == _generation) setState(() => _busy = false);
    }
  }

  @override
  Widget build(BuildContext context) => RefreshIndicator(
    onRefresh: _load,
    child: ListView(
      shrinkWrap: widget.embedded,
      padding: widget.embedded
          ? EdgeInsets.zero
          : const EdgeInsets.all(Ops.padding),
      physics: widget.embedded
          ? const NeverScrollableScrollPhysics()
          : const AlwaysScrollableScrollPhysics(),
      children: [
        if (_busy)
          const LinearProgressIndicator(semanticsLabel: 'Loading server data'),
        if (!_busy && _error != null) OpsError(_error!, retry: _load),
        if (!_busy && _error == null && _data != null)
          widget.builder(context, _data),
      ],
    ),
  );
}

class _VehicleCategoryDialog extends StatefulWidget {
  final ApiService api;
  final Map<String, dynamic>? initial;
  const _VehicleCategoryDialog({required this.api, this.initial});

  @override
  State<_VehicleCategoryDialog> createState() => _VehicleCategoryDialogState();
}

class _VehicleCategoryDialogState extends State<_VehicleCategoryDialog> {
  final _customName = TextEditingController();
  late Future<dynamic> _catalog;
  String? _sector, _type, _error;
  bool _creating = false, _addingCustomType = false;

  @override
  void initState() {
    super.initState();
    _sector = widget.initial?['sector'] as String?;
    _type = widget.initial?['category'] as String?;
    _city = widget.initial?['operatingCityId'] as String?;
    _catalog = widget.api.request('GET', '/provider/vehicle-catalog');
  }

  @override
  void dispose() {
    _customName.dispose();
    super.dispose();
  }

  Future<void> _create() async {
    if (_sector == null ||
        !VehicleCategoryPolicy.isSelectableSector(_sector) ||
        _sector == 'logistics' ||
        _customName.text.trim().isEmpty) {
      return;
    }
    setState(() {
      _creating = true;
      _error = null;
    });
    try {
      final created = await widget.api.request(
        'POST',
        '/provider/vehicle-catalog/custom',
        body: {'sector': _sector, 'name': _customName.text.trim()},
      ) as Map;
      if (!mounted) return;
      setState(() {
        _type = created['code'] as String;
        _customName.clear();
        _addingCustomType = false;
        _catalog = widget.api.request('GET', '/provider/vehicle-catalog');
      });
    } catch (e) {
      developer.log(
        'Custom vehicle type creation failed',
        name: 'VehicleCategoryDialog',
        error: e,
      );
      if (mounted) {
        setState(() => _error = 'Could not create this type. Please retry.');
      }
    } finally {
      if (mounted) setState(() => _creating = false);
    }
  }

  @override
  Widget build(BuildContext context) => AlertDialog(
    title: const Text('Vehicle category and type'),
    content: FutureBuilder<dynamic>(
      future: _catalog,
      builder: (context, snapshot) {
        if (snapshot.connectionState != ConnectionState.done) {
          return const SizedBox(
            height: 80,
            child: Center(child: CircularProgressIndicator()),
          );
        }
        if (snapshot.hasError) {
          return Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Text('Vehicle configuration could not be loaded.'),
              TextButton(
                onPressed: () => setState(
                  () => _catalog = widget.api.request(
                    'GET',
                    '/provider/vehicle-catalog',
                  ),
                ),
                child: const Text('Retry'),
              ),
            ],
          );
        }

        final data = snapshot.data as Map;
        final sectors = List<String>.from(data['sectors'] as List)
            .where(VehicleCategoryPolicy.isSelectableSector)
            .toList();
        final matches = (data['types'] as List)
            .cast<Map>()
            .where(
              (type) =>
                  (type['sector'] == _sector &&
                      VehicleCategoryPolicy.allowsType(
                        _sector,
                        type['name']?.toString(),
                      )) ||
                  (_sector == 'rental' &&
                      type['sector'] == 'premium' &&
                      VehicleCategoryPolicy.allowsType(
                        _sector,
                        type['name']?.toString(),
                      )),
            )
            .toList();
        final types = <String, Map>{
          for (final type in matches) type['code'] as String: type,
        }.values.toList();
        final selectedType = types.any((type) => type['code'] == _type)
            ? _type
            : null;

        return SizedBox(
          width: 420,
          child: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                DropdownButtonFormField<String>(
                  initialValue:
                      sectors.any(
                        (sector) =>
                            sector.toLowerCase() == _sector?.toLowerCase(),
                      )
                      ? _sector
                      : null,
                  decoration: const InputDecoration(
                    labelText: 'Vehicle category',
                  ),
                  items: sectors
                      .map(
                        (sector) => DropdownMenuItem(
                          value: sector,
                          child: Text(sector),
                        ),
                      )
                      .toList(),
                  onChanged: (value) => setState(() {
                    _sector = value;
                    _type = null;
                    _addingCustomType = false;
                    _customName.clear();
                  }),
                ),
                const SizedBox(height: 16),
                DropdownButtonFormField<String>(
                  key: ValueKey('$_sector:$_type'),
                  initialValue: selectedType,
                  isExpanded: true,
                  decoration: const InputDecoration(
                    labelText: 'Configured vehicle type',
                  ),
                  items: types
                      .map(
                        (type) => DropdownMenuItem(
                          value: type['code'] as String,
                          child: Text(type['name'].toString()),
                        ),
                      )
                      .toList(),
                  onChanged: (value) => setState(() => _type = value),
                ),
                if (_sector != null && types.isEmpty)
                  const Text(
                    'No configured types are available for this category.',
                  ),
                if (_sector == 'premium') ...[
                  const SizedBox(height: 16),

                  ),
                  if (cities.isEmpty)
